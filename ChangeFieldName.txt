-- Change field name and make a field mandatory (yellow) --

-- Change field name --

1. NAVIGATE TO HUMAN RESOURCES VISION ENTERPRISE --> ENTER AND MAINTAIN 
2. CLICK THE FIELD NAME : STATUS --> HELP --> DIAGNOSTIC --> EXAMINE 
	BLOCK: PERSON
	FIELD: D_MARITAL_STATUS
3. HELP --> DIAGNOSTIC --> CUSTOM CODE --> PERSONALIZE 
	SEQ	DESC
	1	CHANGING FIELD NAME
4. FROM ACTIONS TAB 
	SEQ
	1	TARGET OBJECT: PERSON.D_MARITAL_STATUS
5. VALIDATE --> APPLY NOW --> SAVE --> CLOSE THE FORM
# STATUS FIELD CHANGED TO MARITAL STATUS


-- make a field mandatory --

1. KNOW FIELD NAME: PERSON.D_MARITAL_STATUS
2. HELP --> DIAGNOSTICS --> CUSTOM CODE --> PERSONALISE 
	SEQ	DESC
	1	MAKING A FIELD MANDATE
3. ACTIONS 
	OBJECT NAME: PERSON.D_MARITAL_STATUS
	PROPERTY: REQUIRED
	VALUE: TRUE
4. VALIDATE --> APPLY NOW --> SAVE
# MARITAL STATUS FIELD IS YELLOW
