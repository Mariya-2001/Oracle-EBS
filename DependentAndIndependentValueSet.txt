-- VALUE SET --

1. APPLICATION DEVELOPER --> APPLICATION --> VALIDATION --> SET
2. WRITE 
	VALUE SET NAME: MARIYA_VS_INDPN
	DESCRIPTION: THIS IS INDEPENDENT VS BASED ON COUNTRY
	LIST TYPE: CHAR 
	SECURITY: NO SECURITY
	MAXIMUM SIZE: 20
	VALIDATION TYPE: INDEPENDENT
3. SAVE

4. WRITE 
	VALUE SET NAME: MARIYA_VS_DEP1
	DESCRIPTION: THIS IS DEPENDENT VS
	LIST TYPE: CHAR 
	SECURITY: NO SECURITY
	MAXIMUM SIZE: 20
	VALIDATION TYPE: DEPENDENT
5. SAVE

6. WRITE 
	VALUE SET NAME: MARIYA_VS_DEP2
	DESCRIPTION: THIS IS DEPENDENT VS
	LIST TYPE: CHAR 
	SECURITY: NO SECURITY
	MAXIMUM SIZE: 20
	VALIDATION TYPE: DEPENDENT
7. SAVE
