-- Factorial Calculation with Exception Handling -- 

Write a PL/SQL program that accepts a input number and returns its factorial as output.
For example, given an input of 5, the program should return 120 (5×4×3×2×1) with handling all the exceptions.

Input 
Factorial of 5 = 5*4*3*2*1 = 120


SQL> CREATE OR REPLACE PROCEDURE MARIYA_EBS_FACTO(ERRBUF OUT VARCHAR2, RETCODE OUT NUMBER, N  NUMBER
) IS 
  2  
  3  S NUMBER := N;  
  4  RES NUMBER := N; 
  5  FACT_1  EXCEPTION; 
  6  FACT_0 EXCEPTION; 
  7  BEGIN 
  8  IF S = 0 OR S = 1 THEN 
  9     RAISE FACT_1; 
 10  ELSIF S < 0 THEN 
 11     RAISE FACT_0; 
 12  END IF; 
 13  LOOP 
 14  S := S - 1; 
 15  RES  := RES * S; 
 16  EXIT WHEN S = 2; 
 17  END LOOP; 
 18  FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' FACTORIAL = '||RES); 
 19  EXCEPTION 
 20  WHEN FACT_1 THEN 
 21  FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' FACTORIAL = 1 '); 
 22  WHEN FACT_0 THEN 
 23  FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' INPUT N CANNOT BE LESS THAN 0 '); 
 24  WHEN NO_DATA_FOUND THEN 
 25  FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' INPUT A NUMBER TO COMPUTE FACTORIAL'); 
 26  WHEN OTHERS THEN 
 27  FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' CHECK YOUR PROGRAM !!! '); 
 28  END;
