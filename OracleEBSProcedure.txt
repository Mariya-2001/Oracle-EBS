-- Create an Oracle EBS procedure & register it in EBS --

1. WRITE THE PROCEDURE IN SQL PLUS
CREATE OR REPLACE PROCEDURE MARIYA_PR_EBS(ERRBUF OUT VARCHAR2, RETCODE OUT NUMBER) IS		 
BEGIN 
FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ‘ WELCOME TO ORACLE EBS PROC..!’); 
END; 
/ 
Procedure Created 


2.REGISTER THE PROCEDURE IN ORACLE EBS 
	-APPLICATION DEVELOPER --> CONCURRENT --> EXECUTABLE
	Name: MARIYA_PR_EBS 
	Short Name: MARIYA_PR_EBS 
	Execution Method: PL/SQL Stored Procedure 
	Execution File Name: MARIYA_PR_EBS 
<SAVE> 

3.VIEW --> QUERY BY EXAMPLE --> ENTER 
	Name: MARIYA_PR_EBS 
4.VIEW --> QUERY BY EXAMPLE --> RUN
    -- SEE YOUR PROCEDURE DETAILS 

 
3. CONCURRENT PROGRAM --> 
	Name: MARIYA_PR_EBS 
	Short Name: MARIYA_PR_EBS 
	Application: AOL --> HIT <TAB> BUTTON
	Executable: MARIYA_PR_EBS 
	Output: HTML 
<SAVE> 
4. SYSTEM ADMINISTRATOR --> SECURITY --> RESBONSIBILITY --> REQUEST 
	Name: MARIYA_RG_PGMS 
	Application: AOL --> HIT <TAB> BUTTON 
	Program: MARIYA_PR_EBS 
<SAVE> 

5. CREATE A NEW USER RESPONSIBILITY (MARIYA_RESP)
	SYSTEM ADMINISTRATOR --> SECURITY --> RESBONSIBILITY --> DEFINE
<SAVE> 

6.TO EXECUTE A PROCEDURE USING THE RESPONSIBILITY 
	-SWITCH TO THE RESPONSIBILITY YOU WANT TO EXECUTE ( MARIYA_RESP)
	-VIEW --> REQUEST
	-SUBMIT A NEW REQUEST -->SINGLE --> OK 
	-CLICK THE LOV
	-MARIYA_PR_EBS 
	-REFRESH DATA
	-COMPLETED 
	-View Output
